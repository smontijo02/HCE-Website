        function escapeHtml(value) {
            return String(value ?? "")
                .replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;")
                .replace(/\"/g, "&quot;")
                .replace(/'/g, "&#39;");
        }

        function setNowPlayingFromRow(row, idx) {
            if (!row) return;
            const song = idx.song >= 0 ? (row[idx.song] || "").trim() : "";
            const artist = idx.artist >= 0 ? (row[idx.artist] || "").trim() : "";
            const requester = idx.requester >= 0 ? (row[idx.requester] || "").trim() : "";

        function isTruthyLiveFlag(value) {
            return /^(yes|y|true|1|live|on)$/i.test(String(value || "").trim());
        }

        function findNowPlayingRow(allRows, idx) {
            if (!allRows.length) return null;

            if (idx.onAir >= 0) {
                const markedLive = [...allRows].reverse().find(r => isTruthyLiveFlag(r[idx.onAir]));
                if (markedLive) return markedLive;
            }

            if (idx.played >= 0) {
                const pending = allRows.filter(r => !r[idx.played] || r[idx.played].trim() === "");
                // Keep fallback in FIFO order to match the visible queue.
                if (pending.length) return pending[0];
            }

            return allRows[allRows.length - 1];
        }

        function parseCSV(text) {
            const rows = [];
            let row = [], field = "", inQuotes = false;
            for (let i = 0; i < text.length; i++) {
                const c = text[i];
                if (inQuotes) {
                    if (c === '"' && text[i + 1] === '"') { field += '"'; i++; }
                    else if (c === '"') { inQuotes = false; }
                    else { field += c; }
                } else {
                    if (c === '"') { inQuotes = true; }
                    else if (c === ',') { row.push(field); field = ""; }
                    else if (c === '\n' || c === '\r') {
                        if (field.length || row.length) { row.push(field); rows.push(row); }
                        field = ""; row = [];
                        if (c === '\r' && text[i + 1] === '\n') i++;
                    } else { field += c; }
                }
            }
            if (field.length || row.length) { row.push(field); rows.push(row); }
            return rows.filter(r => r.length && r.some(cell => cell.trim() !== ""));
        }

        const AUTO_HIDE_ABOVE_ONAIR = true; // set to false if you don't strictly play requests in sheet order

        async function loadQueue() {
            const container = document.getElementById('queue-container');
            const countEl = document.getElementById('queue-count');
            if (!QUEUE_CSV_URL || QUEUE_CSV_URL.startsWith("PASTE_")) {
                container.innerHTML = '<p class="queue-status">Queue display not connected yet — add your published sheet link in the code.</p>';
                if (countEl) countEl.textContent = 'Queue count unavailable';
                return;
            }
            try {
                const sameSource = NOW_PLAYING_CSV_URL === QUEUE_CSV_URL;
                const queueUrl = QUEUE_CSV_URL + (QUEUE_CSV_URL.includes('?') ? '&' : '?') + 'cachebust=' + Date.now();
                const nowPlayingUrl = NOW_PLAYING_CSV_URL + (NOW_PLAYING_CSV_URL.includes('?') ? '&' : '?') + 'cachebust=' + Date.now();
                const [queueRes, nowPlayingRes] = await Promise.all([
                    fetch(queueUrl),
                    sameSource ? Promise.resolve(null) : fetch(nowPlayingUrl)
                ]);

                if (!queueRes.ok) throw new Error('Queue fetch failed: ' + queueRes.status);
                const queueText = await queueRes.text();
                const rows = parseCSV(queueText);
                if (rows.length < 2) {
                    container.innerHTML = '<p class="queue-status">No requests yet — be the first!</p>';
                    if (countEl) countEl.textContent = '0 songs displayed';
                    return;
                }
                const header = rows[0].map(h => h.trim());
                const idxRequester = header.indexOf(COLUMN_MAP.requester);
                const idxSong = header.indexOf(COLUMN_MAP.song);
                const idxArtist = header.indexOf(COLUMN_MAP.artist);
                const idxLink = header.indexOf(COLUMN_MAP.link);
                const idxPlayed = header.indexOf(COLUMN_MAP.played);
                const idxOnAir = header.indexOf(COLUMN_MAP.onAir);
                const idxDedication = header.indexOf(COLUMN_MAP.dedication);

                const nowPlayingRows = sameSource
                    ? rows
                    : (nowPlayingRes && nowPlayingRes.ok ? parseCSV(await nowPlayingRes.text()) : rows);
                if (nowPlayingRows.length >= 2) {
                    const npHeader = nowPlayingRows[0].map(h => h.trim());
                    const npIdx = {
                        requester: npHeader.indexOf(COLUMN_MAP.requester),
                        song: npHeader.indexOf(COLUMN_MAP.song),
                        artist: npHeader.indexOf(COLUMN_MAP.artist),
                        played: npHeader.indexOf(COLUMN_MAP.played),
                        onAir: npHeader.indexOf(COLUMN_MAP.onAir),
                        dedication: npHeader.indexOf(COLUMN_MAP.dedication)
                    };
                    const nowRow = findNowPlayingRow(nowPlayingRows.slice(1), npIdx);
                    setNowPlayingFromRow(nowRow, npIdx);
                }

                // 1. Identify the index of the currently playing live track.
                let currentLiveIdx = -1;
                if (idxOnAir >= 0) {
                    // Find the last row where On Air is flagged true.
                    for (let i = rows.length - 1; i >= 1; i--) {
                        if (isTruthyLiveFlag(rows[i][idxOnAir])) {
                            currentLiveIdx = i;
                            break;
                        }
                    }
                }

                let dataRows = rows.slice(1);

                // 2. High-fidelity queue filtering.
                dataRows = dataRows.filter((r, index) => {
                    const actualRowIdx = index + 1; // Account for header row.

                    // Rule A: Explicitly played rows are hidden.
                    if (idxPlayed >= 0 && r[idxPlayed] && r[idxPlayed].trim() !== "") {
                        return false;
                    }

                    // Rule B: The currently live track is hidden from the queue.
                    if (idxOnAir >= 0 && isTruthyLiveFlag(r[idxOnAir])) {
                        return false;
                    }

                    // Rule C: Any row above the current live row is treated as past/played.
                    if (AUTO_HIDE_ABOVE_ONAIR && currentLiveIdx !== -1 && actualRowIdx < currentLiveIdx) {
                        return false;
                    }

                    return true;
                });
                if (dataRows.length === 0) {
                    container.innerHTML = '<p class="queue-status">No requests yet — be the first!</p>';
                    if (countEl) countEl.textContent = '0 songs displayed';
                    return;
                }
                const maxDisplayed = 15;
                const totalPending = dataRows.length;
                dataRows = dataRows.slice(-maxDisplayed); // most recent pending requests, capped for display

                const displayedCount = dataRows.length;
                const moreWaiting = Math.max(totalPending - displayedCount, 0);
                if (countEl) {
                    countEl.textContent = `${displayedCount} songs displayed${moreWaiting > 0 ? ` • ${moreWaiting} more waiting` : ''}`;
                }

                container.innerHTML = dataRows.map((r, i) => {
                    const requester = escapeHtml(idxRequester >= 0 ? r[idxRequester] : "Listener");
                    const rawSong = idxSong >= 0 ? String(r[idxSong] || "").trim() : "";
                    const rawArtist = idxArtist >= 0 ? String(r[idxArtist] || "").trim() : "";
                    const quotedSong = /^".*"$/.test(rawSong) ? rawSong : `"${rawSong}"`;
                    const song = escapeHtml(quotedSong);
                    const artist = escapeHtml(rawArtist);
                    const rawLink = idxLink >= 0 ? (r[idxLink] || "").trim() : "";
                    const isSafeLink = /^https?:\/\//i.test(rawLink);
                    const songLabel = `${song}${artist ? ' — ' + artist : ''}`;
                    const songDisplay = isSafeLink
                        ? `<a href="${escapeHtml(rawLink)}" target="_blank" rel="noopener noreferrer" class="queue-song queue-link">${songLabel}</a>`
                        : `<span class="queue-song">${songLabel}</span>`;
                    return `<div class="queue-row">
                        <span class="queue-num">${i + 1}.</span>
                        <span class="queue-requester">${requester}</span>
                        ${songDisplay}
                    </div>`;
                }).join('');
            } catch (err) {
                container.innerHTML = '<p class="queue-status">Couldn\'t load the queue right now — try refreshing.</p>';
                if (countEl) countEl.textContent = 'Queue count unavailable';
                console.error(err);
            }
        }

        wireBroadcastButtons();
        loadConfig();
        loadQueue();
        setInterval(loadQueue, 30000);   // refresh queue every 30 seconds
        setInterval(loadConfig, 30000);  // refresh links every 30 seconds
    </script>
</body>
</html>
